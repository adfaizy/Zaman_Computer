function rsBlocks(version, ec){
  const groups = RS_BLOCKS[version - 1][EC_SLOT[ec]];
  const blocks = [];
  for (const [count, total, data] of groups)
    for (let i = 0; i < count; i++) blocks.push({ total, data });
  return blocks;
}

const EXP = new Array(512).fill(0), LOG = new Array(256).fill(0);
EXP[0] = 1;
for (let i = 1; i < 8; i++) EXP[i] = EXP[i - 1] << 1;
for (let i = 8; i < 256; i++) EXP[i] = EXP[i - 4] ^ EXP[i - 5] ^ EXP[i - 6] ^ EXP[i - 8];
for (let i = 0; i < 255; i++) LOG[EXP[i]] = i;
const gexp = n => EXP[n % 255], glog = n => LOG[n];

class Poly {
  constructor(num, shift){
    let offset = 0;
    while (offset < num.length && num[offset] === 0) offset++;
    this.num = num.slice(offset).concat(new Array(shift).fill(0));
  }
  get length(){ return this.num.length; }
  mul(other){
    const num = new Array(this.length + other.length - 1).fill(0);
    for (let i = 0; i < this.length; i++)
      for (let j = 0; j < other.length; j++)
        num[i + j] ^= gexp(glog(this.num[i]) + glog(other.num[j]));
    return new Poly(num, 0);
  }
  mod(other){
    const difference = this.length - other.length;
    if (difference < 0) return this;
    const ratio = glog(this.num[0]) - glog(other.num[0]);
    const num = new Array(other.length);
    for (let i = 0; i < other.length; i++) num[i] = this.num[i] ^ gexp(glog(other.num[i]) + ratio);
    return new Poly(num.concat(this.num.slice(other.length)), 0).mod(other);
  }
}

class BitBuffer {
  constructor(){ this.buffer = []; this.length = 0; }
  put(num, len){ for (let i = len - 1; i >= 0; i--) this.putBit(((num >>> i) & 1) === 1); }
  putBit(bit){
    const idx = Math.floor(this.length / 8);
    if (this.buffer.length <= idx) this.buffer.push(0);
    if (bit) this.buffer[idx] |= 0x80 >>> (this.length % 8);
    this.length++;
  }
}

function createBytes(buffer, rsBlks){
  let offset = 0, maxDc = 0, maxEc = 0;
  const dcdata = [], ecdata = [];
  for (const blk of rsBlks){
    const dcCount = blk.data, ecCount = blk.total - blk.data;
    maxDc = Math.max(maxDc, dcCount); maxEc = Math.max(maxEc, ecCount);
    const currentDc = buffer.slice(offset, offset + dcCount).map(b => b & 0xff); offset += dcCount;
    let rsPoly = new Poly([1], 0);
    for (let i = 0; i < ecCount; i++) rsPoly = rsPoly.mul(new Poly([1, gexp(i)], 0));
    const raw = new Poly(currentDc, rsPoly.length - 1);
    const mod = raw.mod(rsPoly);
    const currentEc = new Array(ecCount).fill(0);
    const modOffset = mod.num.length - ecCount;
    for (let i = 0; i < ecCount; i++){ const idx = i + modOffset; currentEc[i] = idx >= 0 ? mod.num[idx] : 0; }
    dcdata.push(currentDc); ecdata.push(currentEc);
  }
  const data = [];
  for (let i = 0; i < maxDc; i++) for (const dc of dcdata) if (i < dc.length) data.push(dc[i]);
  for (let i = 0; i < maxEc; i++) for (const ec of ecdata) if (i < ec.length) data.push(ec[i]);
  return data;
}

function createData(version, ec, dataBytes){
  const buffer = new BitBuffer();
  buffer.put(4, 4);                                   /* mode: byte */
  buffer.put(dataBytes.length, modeByteBits(version));
  for (const b of dataBytes) buffer.put(b, 8);
  const blks = rsBlocks(version, ec);
  const bitLimit = blks.reduce((s, b) => s + b.data * 8, 0);
  for (let i = 0; i < Math.min(bitLimit - buffer.length, 4); i++) buffer.putBit(false);
  while (buffer.length % 8) buffer.putBit(false);
  const bytesToFill = (bitLimit - buffer.length) / 8;
  for (let i = 0; i < bytesToFill; i++) buffer.put(i % 2 === 0 ? PAD0 : PAD1, 8);
  return createBytes(buffer.buffer, blks);
}

const bitLen = n => { let d = 0; while (n !== 0){ d++; n >>>= 1; } return d; };
const G15 = (1 << 10) | (1 << 8) | (1 << 5) | (1 << 4) | (1 << 2) | (1 << 1) | (1 << 0);
const G18 = (1 << 12) | (1 << 11) | (1 << 10) | (1 << 9) | (1 << 8) | (1 << 5) | (1 << 2) | (1 << 0);
const G15_MASK = (1 << 14) | (1 << 12) | (1 << 10) | (1 << 4) | (1 << 1);
function bchTypeInfo(data){
  let d = data << 10;
  while (bitLen(d) - bitLen(G15) >= 0) d ^= G15 << (bitLen(d) - bitLen(G15));
  return ((data << 10) | d) ^ G15_MASK;
}
function bchTypeNumber(data){
  let d = data << 12;
  while (bitLen(d) - bitLen(G18) >= 0) d ^= G18 << (bitLen(d) - bitLen(G18));
  return (data << 12) | d;
}
const MASK_FNS = [
  (i, j) => (i + j) % 2 === 0, (i, j) => i % 2 === 0, (i, j) => j % 3 === 0, (i, j) => (i + j) % 3 === 0,
  (i, j) => (Math.floor(i / 2) + Math.floor(j / 3)) % 2 === 0, (i, j) => (i * j) % 2 + (i * j) % 3 === 0,
  (i, j) => ((i * j) % 2 + (i * j) % 3) % 2 === 0, (i, j) => ((i * j) % 3 + (i + j) % 2) % 2 === 0
];
function buildBlank(version){
  const size = version * 4 + 17;
  const m = Array.from({ length: size }, () => new Array(size).fill(null));
  const probe = (row, col) => {
    for (let r = -1; r < 8; r++) for (let c = -1; c < 8; c++){
      const rr = row + r, cc = col + c;
      if (rr < 0 || rr >= size || cc < 0 || cc >= size) continue;
      m[rr][cc] = (r >= 0 && r <= 6 && (c === 0 || c === 6)) ||
                  (c >= 0 && c <= 6 && (r === 0 || r === 6)) ||
                  (r >= 2 && r <= 4 && c >= 2 && c <= 4);
    }
  };
  probe(0, 0); probe(0, size - 7); probe(size - 7, 0);
  const pos = PATTERN_POSITION[version - 1];
  for (const r of pos) for (const c of pos){
    if (m[r][c] !== null) continue;
    for (let dr = -2; dr <= 2; dr++) for (let dc = -2; dc <= 2; dc++){
      if (Math.abs(dr) === 2 || Math.abs(dc) === 2 || (dr === 0 && dc === 0)) m[r + dr][c + dc] = true; else m[r + dr][c + dc] = false;
    }
  }
  for (let i = 8; i < size - 8; i++){
    if (m[i][6] === null) m[i][6] = i % 2 === 0;
    if (m[6][i] === null) m[6][i] = i % 2 === 0;
  }
  return m;
}
function setupTypeInfo(m, ec, maskPattern, test){
  const size = m.length;
  const bits = bchTypeInfo((ec << 3) | maskPattern);
  const bit = i => !test && ((bits >> i) & 1) === 1;
  for (let i = 0; i < 15; i++){
    const mod = bit(i);
    if (i < 6) m[i][8] = mod; else if (i < 8) m[i + 1][8] = mod; else m[size - 15 + i][8] = mod;
  }
  for (let i = 0; i < 15; i++){
    const mod = bit(i);
    if (i < 8) m[8][size - i - 1] = mod; else if (i === 8) m[8][7] = mod; else m[8][14 - i] = mod;
  }
  m[size - 8][8] = !test;
}
function setupTypeNumber(m, version, test){
  const size = m.length;
  const bits = bchTypeNumber(version);
  for (let i = 0; i < 18; i++){
    const mod = !test && ((bits >> i) & 1) === 1;
    m[Math.floor(i / 3)][i % 3 + size - 8 - 3] = mod;
  }
  for (let i = 0; i < 18; i++){
    const mod = !test && ((bits >> i) & 1) === 1;
    m[i % 3 + size - 8 - 3][Math.floor(i / 3)] = mod;
  }
}
function mapData(m, data, maskPattern){
  const size = m.length;
  const maskFn = MASK_FNS[maskPattern];
  let inc = -1, row = size - 1, bitIndex = 7, byteIndex = 0;
  for (let col = size - 1; col > 0; col -= 2){
    if (col <= 6) col--;
    for (;;){
      for (const c of [col, col - 1]){
        if (m[row][c] === null){
          let dark = false;
          if (byteIndex < data.length) dark = ((data[byteIndex] >> bitIndex) & 1) === 1;
          if (maskFn(row, c)) dark = !dark;
          m[row][c] = dark;
          bitIndex--;
          if (bitIndex === -1){ byteIndex++; bitIndex = 7; }
        }
      }
      row += inc;
      if (row < 0 || row >= size){ row -= inc; inc = -inc; break; }
    }
  }
}
function buildModules(version, ec, dataCodewords, maskPattern, test){
  const m = buildBlank(version);
  setupTypeInfo(m, ec, maskPattern, test);
  if (version >= 7) setupTypeNumber(m, version, test);
  mapData(m, dataCodewords, maskPattern);
  return m;
}
function lostPoint(m){
  const n = m.length;
  let lost = 0;
  /* level 1: runs of 5+ same color */
  for (let r = 0; r < n; r++){
    let run = 1;
    for (let c = 1; c < n; c++){
      if (m[r][c] === m[r][c - 1]) run++;
      else { if (run >= 5) lost += run - 2; run = 1; }
    }
    if (run >= 5) lost += run - 2;
  }
  for (let c = 0; c < n; c++){
    let run = 1;
    for (let r = 1; r < n; r++){
      if (m[r][c] === m[r - 1][c]) run++;
      else { if (run >= 5) lost += run - 2; run = 1; }
    }
    if (run >= 5) lost += run - 2;
  }
  /* level 2: 2x2 blocks */
  for (let r = 0; r < n - 1; r++) for (let c = 0; c < n - 1; c++){
    const v = m[r][c];
    if (v === m[r][c + 1] && v === m[r + 1][c] && v === m[r + 1][c + 1]) lost += 3;
  }
  /* level 3: finder-like 1:1:3:1:1 patterns with quiet zone */
  const P1 = [true, false, true, true, true, false, true, false, false, false, false];
  const P2 = [false, false, false, false, true, false, true, true, true, false, true];
  for (let r = 0; r < n; r++) for (let c = 0; c + 10 < n; c++){
    const seg = [m[r][c], m[r][c+1], m[r][c+2], m[r][c+3], m[r][c+4], m[r][c+5], m[r][c+6], m[r][c+7], m[r][c+8], m[r][c+9], m[r][c+10]];
    if (seg.every((v, i) => v === P1[i]) || seg.every((v, i) => v === P2[i])) lost += 40;
  }
  for (let c = 0; c < n; c++) for (let r = 0; r + 10 < n; r++){
    const seg = [m[r][c], m[r+1][c], m[r+2][c], m[r+3][c], m[r+4][c], m[r+5][c], m[r+6][c], m[r+7][c], m[r+8][c], m[r+9][c], m[r+10][c]];
    if (seg.every((v, i) => v === P1[i]) || seg.every((v, i) => v === P2[i])) lost += 40;
  }
  /* level 4: dark module proportion */
  const dark = m.flat().filter(Boolean).length;
  lost += Math.floor(Math.abs(dark * 100 / (n * n) - 50) / 5) * 10;
  return lost;
}
function pickVersion(ec, byteLen){
  for (let v = 1; v <= 40; v++){
    if (BIT_LIMIT[ec][v] >= 4 + modeByteBits(v) + byteLen * 8) return v;
  }
  throw new Error('QR data too long');
}
function qrMatrix(text, ec){
  ec = ec === undefined ? EC_M : ec;
  const dataBytes = Array.from(new TextEncoder().encode(String(text)));
  if (!dataBytes.length || dataBytes.length > 2953) throw new Error('QR: bad length');
  const version = pickVersion(ec, dataBytes.length);
  const dataCodewords = createData(version, ec, dataBytes);
  let bestMask = 0, bestLost = Infinity;
  for (let i = 0; i < 8; i++){
    const test = buildModules(version, ec, dataCodewords, i, true);
    const lost = lostPoint(test);
    if (i === 0 || bestLost > lost){ bestLost = lost; bestMask = i; }
  }
  return buildModules(version, ec, dataCodewords, bestMask, false);
}
function drawQR(canvas, text, opts){
  opts = opts || {};
  const module = opts.module || 4, border = 4 * module;
  let matrix;
  try { matrix = qrMatrix(text, EC_M); } catch (e) { return false; }
  const n = matrix.length;
  canvas.width = n * module + border * 2;
  canvas.height = n * module + border * 2;
  const ctx = canvas.getContext('2d');
  ctx.fillStyle = '#ffffff'; ctx.fillRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = '#111111';
  for (let r = 0; r < n; r++) for (let c = 0; c < n; c++) if (matrix[r][c]) ctx.fillRect(border + c * module, border + r * module, module, module);
  return true;
}

/* Amount in words (Pakistani numbering: lakh / crore) */
const ONES_W = ['','One','Two','Three','Four','Five','Six','Seven','Eight','Nine','Ten','Eleven','Twelve','Thirteen','Fourteen','Fifteen','Sixteen','Seventeen','Eighteen','Nineteen'];
const TENS_W = ['','','Twenty','Thirty','Forty','Fifty','Sixty','Seventy','Eighty','Ninety'];
const twoW = n => n < 20 ? ONES_W[n] : TENS_W[Math.floor(n / 10)] + (n % 10 ? ' ' + ONES_W[n % 10] : '');
const threeW = n => { const h = Math.floor(n / 100), r = n % 100; return (h ? ONES_W[h] + ' Hundred' + (r ? ' ' : '') : '') + (r ? twoW(r) : ''); };
function amountToWords(n){
  n = Math.floor(Math.abs(Number(n) || 0));
  if (!n) return 'Zero';
  const crore = Math.floor(n / 1e7), lakh = Math.floor((n % 1e7) / 1e5), thou = Math.floor((n % 1e5) / 1e3), rem = n % 1000;
  let s = '';
  if (crore) s += threeW(crore) + ' Crore ';
  if (lakh)  s += threeW(lakh) + ' Lakh ';
  if (thou)  s += threeW(thou) + ' Thousand ';
  if (rem)   s += threeW(rem);
  return s.trim();
}

/* ───────── 17B. PDF INVOICE — generated in-browser, no libraries ─────────
   Produces a vector A4 PDF identical in layout to the printed invoice:
   black/gold header band, FBR/NTN strip, item table, QR code, totals,
   amount in words (lakh/crore). Fully offline-capable.            */
const PDFW = 595.28, PDFH = 841.89, PDFML = 40, PDFMR = PDFW - 40;
const INK = [17,17,17], GOLD = [212,175,55], GOLD_L = [240,214,117], GOLD_D = [138,109,31];
const CREAM = [251,249,241], GREYT = [88,88,100], LGREY = [201,201,209], PAPERW = [253,252,248];
const measCtx = document.createElement('canvas').getContext('2d');
const twPt = (s, size, bold) => { measCtx.font = (bold ? '700 ' : '400 ') + size + 'px Helvetica, Arial, sans-serif'; return measCtx.measureText(String(s)).width; };
const f2 = n => Number(n).toFixed(2);
const rgbStr = c => c.map(v => (v/255).toFixed(3)).join(' ');

function pdfSanitize(s){
  return String(s)
    .replace(/[\u2013\u2014]/g, '-').replace(/[\u2018\u2019\u201A]/g, "'")
    .replace(/[\u201C\u201D\u201E]/g, '"').replace(/\u2026/g, '...')
    .replace(/\u00D7/g, 'x').replace(/[\u00B7\u2022\u25C6\u2605\u2606]/g, '-')
    .replace(/[^\x20-\x7E]/g, '');
}
const pdfEscape = s => pdfSanitize(s).replace(/\\/g, '\\\\').replace(/\(/g, '\\(').replace(/\)/g, '\\)');
const fitText = (s, maxW, size, bold) => {
  let t = pdfSanitize(s);
  if (twPt(t, size, bold) <= maxW) return t;
  while (t.length > 3 && twPt(t + '...', size, bold) > maxW) t = t.slice(0, -1);
  return t + '...';
};

function buildInvoicePdf(){
  const pagesOps = [];
  let ops = [];
  const newPage = () => { ops = []; pagesOps.push(ops); };
  const Y = t => PDFH - t; // top-origin helper

  const fillRect = (x, yTop, w, h, color) => { ops.push(rgbStr(color) + ' rg', f2(x) + ' ' + f2(Y(yTop) - h) + ' ' + f2(w) + ' ' + f2(h) + ' re f'); };
  const strokeRect = (x, yTop, w, h, color, lw) => { ops.push(rgbStr(color) + ' RG', f2(lw || 0.7) + ' w', f2(x) + ' ' + f2(Y(yTop) - h) + ' ' + f2(w) + ' ' + f2(h) + ' re S'); };
  const hline = (yTop, color, lw, x1, x2) => { ops.push(rgbStr(color) + ' RG', f2(lw) + ' w', f2(x1 == null ? PDFML : x1) + ' ' + f2(Y(yTop)) + ' m ' + f2(x2 == null ? PDFMR : x2) + ' ' + f2(Y(yTop)) + ' l S'); };
  const text = (s, x, yBaseTop, size, font, color, charSpace) => {
    ops.push(rgbStr(color || INK) + ' rg');
    ops.push('BT /' + font + ' ' + size + ' Tf ' + (charSpace || 0) + ' Tc 1 0 0 1 ' + f2(x) + ' ' + f2(Y(yBaseTop)) + ' Tm (' + pdfEscape(s) + ') Tj ET');
  };
  const rightText = (s, xRight, yBaseTop, size, font, color) => text(s, xRight - twPt(pdfSanitize(s), size, font === 'F2'), yBaseTop, size, font, color);
  const centerText = (s, yBaseTop, size, font, color) => text(s, (PDFW - twPt(pdfSanitize(s), size, font === 'F2')) / 2, yBaseTop, size, font, color);

  const t = invoiceTotals();
  const d = new Date(invDateEl.value || Date.now());
  const dateStr = d.toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric' });
  const no = invNoEl.value;
  const customer = properCase($('#invCustomer').value) || 'Walk-in Customer';
  const org = $('#invOrg').value.trim();
  const phone = $('#invPhone').value.trim();
  const type = $('#invType').value, payment = $('#invPayment').value;

  /* ───────── PAGE SCAFFOLD ───────── */
  const BAND_H = 132;
  const drawBand = () => {
    fillRect(0, 0, PDFW, BAND_H, INK);
    fillRect(0, BAND_H, PDFW, 2.6, GOLD);
    text('ZAMAN COMPUTER', PDFML, 37, 22.5, 'F2', [255,255,255], 1.1);
    text('PREMIUM  COMPUTER  &  LAPTOP  STORE', PDFML, 53, 5.6, 'F2', GOLD, 1.6);
    text('1, Mohal Street, Nazd Jamia Masjid Al Furqan,', PDFML, 62, 6.9, 'F1', [196,196,203]);
    text('Gulberg Town, Raiwind, Lahore', PDFML, 73, 6.9, 'F1', [196,196,203]);
    text('Proprietor: ' + SHOP.proprietor, PDFML, 84, 6.9, 'F1', [196,196,203]);
    text('NTN: ' + SHOP.ntn + '   |   Ref: ' + SHOP.refNo, PDFML, 95, 6.9, 'F1', [196,196,203]);
    text(SHOP.phoneDisplay + '   |   ' + SHOP.email, PDFML, 106, 6.9, 'F1', [196,196,203]);
    /* invoice number + QR on the right, parallel to the brand block */
    const vMat = qrMatrix(verifyPayload(), QR.EC_M);
    const mSize = 0.82, qs = 4 * mSize;
    const qrSize = vMat.length * mSize, boxSize = qrSize + 2 * qs;
    const bx = PDFMR - boxSize, by = 38;
    fillRect(bx, by, boxSize, boxSize, [255, 255, 255]);
    ops.push(rgbStr(INK) + ' rg');
    for (let r = 0; r < vMat.length; r++) for (let c = 0; c < vMat.length; c++){
      if (vMat[r][c]){
        const x = bx + qs + c * mSize, top = by + qs + r * mSize;
        ops.push(f2(x) + ' ' + f2(Y(top) - mSize) + ' ' + f2(mSize) + ' ' + f2(mSize) + ' re f');
      }
    }
    rightText('INVOICE', PDFMR, by + boxSize + 8, 6.2, 'F2', GOLD_L, 1.6);
    rightText(no, PDFMR, by + boxSize + 16.5, 6.4, 'F2', [255,255,255]);
  };
  const drawSlimHeader = (pageNo) => {
    fillRect(0, 0, PDFW, 30, INK);
    fillRect(0, 30, PDFW, 1.8, GOLD);
    text('ZAMAN COMPUTER  -  INVOICE ' + no + '  (page ' + pageNo + ')', PDFML, 19, 8, 'F2', GOLD);
    rightText(dateStr, PDFMR, 19, 8, 'F1', [225,225,225]);
  };
  const drawTableHeader = (yTop) => {
    fillRect(PDFML, yTop, PDFMR - PDFML, 20, INK);
    text('#', PDFML + 6, yTop + 13.5, 7.5, 'F2', GOLD_L);
    text('DESCRIPTION', 74, yTop + 13.5, 7.5, 'F2', GOLD_L, 0.8);
    rightText('QTY', 331, yTop + 13.5, 7.5, 'F2', GOLD_L, 0.8);
    rightText('RATE', 441, yTop + 13.5, 7.5, 'F2', GOLD_L, 0.8);
    rightText('AMOUNT', 551, yTop + 13.5, 7.5, 'F2', GOLD_L, 0.8);
  };

  newPage();
  drawBand();

  /* ───────── META STRIP (light band under the header) ───────── */
  const META_TOP = 140, META_H = 34;
  fillRect(0, META_TOP, PDFW, META_H, CREAM);
  hline(META_TOP, LGREY, 0.5);
  hline(META_TOP + META_H, LGREY, 0.5);
  const CELL = 99, GAP = 4, CELL_MX = PDFML + 2;
  const metaCells = [
    ['INVOICE NO', no], ['BILL TYPE', type], ['INVOICE DATE', dateStr],
    ['PAYMENT', payment], ['FBR STATUS', 'Income Tax: Active']
  ];
  metaCells.forEach(([k, v], i) => {
    const cx = CELL_MX + i * (CELL + GAP);
    text(k, cx, META_TOP + 12.5, 5.2, 'F2', GOLD_D, 0.9);
    text(fitText(v, CELL - 10, 7.6, true), cx, META_TOP + 24.5, 7.6, 'F2', INK);
    if (i > 0) fillRect(cx - GAP / 2 - 0.25, META_TOP + 7, 0.5, META_H - 14, LGREY);
  });

  /* ───────── BILLED TO / FROM ───────── */
  const pY = META_TOP + META_H + 12;
  fillRect(PDFML, pY, 247, 44, CREAM);
  fillRect(308, pY, 247, 44, CREAM);
  fillRect(PDFML, pY, 3, 44, GOLD);
  fillRect(308, pY, 3, 44, GOLD);
  text('BILLED TO', PDFML + 10, pY + 11, 6, 'F2', GOLD_D, 1.4);
  text(fitText(customer, 225, 9.5, true), PDFML + 10, pY + 23, 9.5, 'F2', INK);
  if (org) text(fitText(org, 225, 6.8, false), PDFML + 10, pY + 33, 6.8, 'F1', GREYT);
  if (phone) text(phone, PDFML + 10, pY + (org ? 41.5 : 35), 6.5, 'F1', GREYT);
  text('FROM', 318, pY + 12, 6, 'F2', GOLD_D, 1.4);
  text(SHOP.name, 318, pY + 25, 9, 'F2', INK);
  text('Gulberg Town, Raiwind, Lahore', 318, pY + 37, 7, 'F1', GREYT);

  /* ───────── TABLE ───────── */
  let yCursor = pY + 56;
  drawTableHeader(yCursor);
  yCursor += 20;
  const rowH = 19.5;
  invItems.forEach((it, i) => {
    if (yCursor + rowH > PDFH - 190) { // move rows continuation to a new page
      newPage();
      drawSlimHeader(pagesOps.length);
      yCursor = 46;
      drawTableHeader(yCursor);
      yCursor += 20;
    }
    if (i % 2 === 0) fillRect(PDFML, yCursor, PDFMR - PDFML, rowH, PAPERW);
    const amt = it.qty * it.rate;
    text(String(i + 1), PDFML + 6, yCursor + 13, 8, 'F1', GREYT);
    text(fitText(it.desc, 220, 7.6, false), 74, yCursor + 13, 7.6, 'F1', INK);
    rightText(String(it.qty), 331, yCursor + 13, 8, 'F1', INK);
    rightText(fmtRs(it.rate), 441, yCursor + 13, 8, 'F1', INK);
    rightText(fmtRs(amt), 551, yCursor + 13, 8, 'F2', INK);
    hline(yCursor + rowH, [228,226,214], 0.5);
    yCursor += rowH;
  });
  if (!invItems.length){ text('(No items on this bill)', PDFML + 10, yCursor + 14, 8, 'F3', GREYT); yCursor += rowH; }

  /* ───────── TOTALS + AMOUNT IN WORDS ───────── */
  if (yCursor + 108 > PDFH - 96) { // not enough room on this page — fresh page for totals
    newPage();
    drawSlimHeader(pagesOps.length);
    yCursor = 46;
  }
  const tx = 320, tw2 = PDFMR - tx;
  const words = ('Rupees ' + amountToWords(t.grand) + ' only').split(' ');
  text('AMOUNT IN WORDS', PDFML, yCursor + 10, 5.6, 'F2', GOLD_D, 1.2);
  let wLine = '', wy = yCursor + 23;
  words.forEach(w => {
    if (twPt(wLine + ' ' + w, 8.6, true) > 250) { text(wLine.trim(), PDFML, wy, 8.6, 'F2', INK); wy += 12; wLine = ''; }
    wLine += ' ' + w;
  });
  text(wLine.trim(), PDFML, wy, 8.6, 'F2', INK);
  text('TERMS: Check items at the counter.', PDFML, wy + 17, 6.2, 'F1', GREYT);
  text('Warranty only with this invoice + stamp.', PDFML, wy + 26, 6.2, 'F1', GREYT);
  text('Goods once sold will not be taken back.', PDFML, wy + 35, 6.2, 'F1', GREYT);
  if ($('#invNotes').value.trim()) text(fitText('Note: ' + $('#invNotes').value.trim(), 250, 6.2, false), PDFML, wy + 45, 6.2, 'F3', GREYT);

  const tRow = (label, value, bold, band) => {
    if (band) fillRect(tx, yCursor, tw2, 21, INK);
    text(label, tx + 8, yCursor + 14, band ? 8.5 : 7.6, band ? 'F2' : 'F1', band ? GOLD_L : GREYT);
    rightText(value, PDFMR - 8, yCursor + 14, band ? 9.5 : 8, band ? 'F2' : 'F1', band ? GOLD_L : INK);
    yCursor += band ? 21 : 16.5;
  };
  tRow('Subtotal', fmtRs(t.subtotal), false, false);
  tRow('Owner Discount', '- ' + fmtRs(t.discount), false, false);
  tRow('Sales Tax (' + Math.round(SHOP.gstRate * 100) + '%)', fmtRs(t.gst), false, false);
  tRow('GRAND TOTAL', fmtRs(t.grand), true, true);

  /* ───────── FBR / NTN FOOTER STRIP ───────── */
  const fy = PDFH - 66;
  hline(fy - 12, GOLD, 1.4);
  centerText('NTN: ' + SHOP.ntn + '  -  FBR - ' + SHOP.fbrStatus + '  -  Reg. ' + SHOP.registeredOn + '  -  ' + SHOP.taxOffice + '  -  ' + SHOP.principals, fy - 1, 6.1, 'F1', GREYT);
  centerText('This is not a sales-tax invoice (business not registered for Sales Tax).', fy + 9, 6.1, 'F3', GREYT);
  centerText('Thank you for your business!  -  Zaman Computer', fy + 19, 6.8, 'F2', GOLD_D);

  /* ───────── ASSEMBLE PDF FILE ───────── */
  const objects = {};
  const setObj = (id, body) => { objects[id] = body; };
  pagesOps.forEach((pageOps, idx) => {
    const cid = 100 + idx * 2, pid = 101 + idx * 2;
    const stream = pageOps.join('\n');
    setObj(cid, '<< /Length ' + stream.length + ' >>\nstream\n' + stream + '\nendstream');
    setObj(pid, '<< /Type /Page /Parent 2 0 R /MediaBox [0 0 ' + f2(PDFW) + ' ' + f2(PDFH) + '] /Resources << /Font << /F1 3 0 R /F2 4 0 R /F3 5 0 R >> >> /Contents ' + cid + ' 0 R >>');
  });
  setObj(1, '<< /Type /Catalog /Pages 2 0 R >>');
  setObj(2, '<< /Type /Pages /Kids [' + pagesOps.map((_, i) => (101 + i * 2) + ' 0 R').join(' ') + '] /Count ' + pagesOps.length + ' >>');
  setObj(3, '<< /Type /Font /Subtype /Type1 /BaseFont /Helvetica >>');
  setObj(4, '<< /Type /Font /Subtype /Type1 /BaseFont /Helvetica-Bold >>');
  setObj(5, '<< /Type /Font /Subtype /Type1 /BaseFont /Helvetica-Oblique >>');

  let pdf = '%PDF-1.4\n';
  const ids = Object.keys(objects).map(Number).sort((a, b) => a - b);
  const maxId = ids[ids.length - 1];
  const offsets = new Array(maxId + 1).fill(0);
  ids.forEach(id => { offsets[id] = pdf.length; pdf += id + ' 0 obj\n' + objects[id] + '\nendobj\n'; });
  const xref = pdf.length;
  pdf += 'xref\n0 ' + (maxId + 1) + '\n';
  pdf += '0000000000 65535 f \n';
  for (let i = 1; i <= maxId; i++) pdf += String(offsets[i] || 0).padStart(10, '0') + ' 00000 n \n';
  pdf += 'trailer\n<< /Size ' + (maxId + 1) + ' /Root 1 0 R >>\nstartxref\n' + xref + '\n%%EOF';
  return pdf;
}

function invoicePdfBlob(){ return new Blob([buildInvoicePdf()], { type: 'application/pdf' }); }
function downloadBlob(blob, filename){
  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob);
  a.download = filename;
  document.body.appendChild(a);
  a.click();
  setTimeout(() => { URL.revokeObjectURL(a.href); a.remove(); }, 4000);
}

  renderInvoice();
  const filename = invoicePdfFilename();

  const customer = properCase($('#invCustomer').value) || 'Walk-in Customer';
  const shareText = 'Zaman Computer — Invoice ' + invNoEl.value + ' — ' + customer + ' — Grand Total: ' + fmtRs(invoiceTotals().grand);
  if (navigator.canShare && navigator.canShare({ files: [file] })) {
    try {
      await navigator.share({ files: [file], title: 'Zaman Computer — Invoice ' + invNoEl.value, text: shareText });
      showToast('Invoice PDF shared via WhatsApp.');
      return;
    } catch (err) {
      if (err && err.name === 'AbortError') return; // user cancelled the share sheet
    }
  }
  /* Desktop / unsupported browsers: download the PDF, then open WhatsApp on the CUSTOMER's number */
  downloadBlob(invoicePdfBlob(), filename);
  const link = waCustomerLink(buildInvoiceWhatsAppText());
  if (!link) {
    showToast('PDF downloaded. Customer mobile invalid — attach it to the customer chat manually.');
    return;
  }
  window.open(link, '_blank', 'noopener');
  showToast('PDF downloaded — attach ' + filename + ' in the customer chat that just opened.');
});
